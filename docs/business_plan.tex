\section{Resumo Executivo (Elevator Pitch)}

\begin{highlightbox}[title=A Oportunidade EquiLock]
O \textbf{EquiLock} é uma solução pioneira no sector equestre que automatiza a abertura e alimentação agendada em cavalariças. Combina um invólucro de hardware inteligente instalado \textbf{exclusivamente no exterior das baias} com uma plataforma cloud em tempo real, eliminando custos excessivos de mão-de-obra e promovendo o bem-estar animal.
\end{highlightbox}

\subsection{O Problema de Mercado}
\begin{itemize}[leftmargin=*]
    \textbf{\item Custos Operacionais Elevados:} A alimentação e soltura matinal dos cavalos exige a presença física de tratadores em horários madrugadores (06:00h--07:00h), representando até 40\% dos custos operacionais de uma cavalariça.
    \textbf{\item Stresse e Saúde Animal:} Atrasos na rotina de alimentação causam ansiedade, stresse comportamental e graves problemas digestivos (ex: cólicas e úlceras gástricas).
    \textbf{\item Ausência de Auditoria:} Donos de cavalos de elevado valor não possuem forma de verificar remotamente se o seu animal foi alimentado ou solto no horário correto.
\end{itemize}

\subsection{A Solução EquiLock}
O EquiLock resolve estes desafios através de uma abordagem dupla:
\begin{enumerate}
    \item \textbf{Módulo de Hardware Exterior Multi-Box:} Um único microcontrolador de baixo custo (ESP32) controla até 4 trincos/atuadores de 12V. O pulso de atuação é rigidamente limitado a \textbf{4 segundos} por firmware para evitar aquecimento, contando com mecanismo de \textbf{abertura manual de emergência}.
    \item \textbf{Plataforma SaaS Multi-Tenant (RBAC):} Painel de controlo web intuitivo com diferenciação de perfis (\texttt{DEVELOPER}, \texttt{CLIENT\_ADMIN} e \texttt{OPERATOR}) e 4 modos de operação (Manual, Sequência Diária com pré-aviso por LED, Agendamento NTP e Fallback Offline).
\end{enumerate}

\section{Arquitetura Técnica Integrada}

\subsection{Hardware IoT (Regras de Segurança Físicas)}
\begin{itemize}[leftmargin=*]
    \item \textbf{Atuação Temporizada de 4 Segundos:} O solenoide de 12V recebe energia unicamente durante 4 segundos exatos, prevenindo o sobreaquecimento e falha de bobina.
    \item \textbf{Resiliência Autónoma Offline:} Se o sinal Wi-Fi falhar, o microcontrolador ESP32 utiliza o seu relógio interno (NTP com software RTC) e EEPROM para manter o cumprimento rigoroso dos agendamentos.
    \item \textbf{LED de Diagnóstico:} Piscar lento (Procura Wi-Fi), Fixo por 3s (Conectado), Piscar rápido por 5s (Aviso prévio de abertura) e Fixo por 4s (Abertura em curso).
\end{itemize}

\subsection{Software SaaS \& Plataforma Cloud (RBAC)}
\begin{itemize}[leftmargin=*]
    \item \textbf{DEVELOPER (Super Admin):} Registo de novos armários/máquinas por UUID/MAC e atribuição a clientes.
    \item \textbf{CLIENT\_ADMIN (Dono do Haras):} Gestão da equipa, associação de tratadores a baias específicas e parametrização de horários.
    \item \textbf{OPERATOR (Tratador):} Acesso simplificado a botões manuais e início de sequência diária.
\end{itemize}
